import Fastify, { type FastifyInstance } from "fastify";
import cors from "@fastify/cors";
import sensible from "@fastify/sensible";
import { createDb } from "@waapi/db";
import {
  serializerCompiler,
  validatorCompiler,
} from "fastify-type-provider-zod";
import { config } from "./config.js";
import authPlugin from "./plugins/auth.js";
import errorHandler from "./plugins/error-handler.js";
import rateLimit from "./plugins/rate-limit.js";
import requestId from "./plugins/request-id.js";
import authRoutes from "./routes/v1/auth.js";
import healthRoutes from "./routes/v1/health.js";
import messageRoutes from "./routes/v1/messages.js";
import sendRoutes from "./routes/v1/send.js";
import sessionRoutes from "./routes/v1/sessions.js";
import webhookRoutes from "./routes/v1/webhooks.js";
import { closeQueues } from "./services/queues.js";
import { purgeAuthArtifacts } from "./services/auth.js";
import "./types.js";

export type BuiltApp = FastifyInstance & {
  registeredRoutes: Array<{ method: string; url: string }>;
};

export async function buildApp(): Promise<BuiltApp> {
  const app = Fastify({
    // Numeric hop count (default 1). Boolean `true` would trust the entire chain.
    trustProxy: config.TRUST_PROXY_HOPS as unknown as boolean,
    logger:
      config.NODE_ENV === "development"
        ? { level: config.LOG_LEVEL, transport: { target: "pino-pretty" } }
        : { level: config.LOG_LEVEL },
    genReqId: (request) => {
      const supplied = request.headers["x-request-id"];
      return typeof supplied === "string" && supplied.length <= 128
        ? supplied
        : crypto.randomUUID();
    },
  }) as unknown as BuiltApp;

  const registeredRoutes: Array<{ method: string; url: string }> = [];
  app.decorate("registeredRoutes", registeredRoutes);
  app.addHook("onRoute", (route) => {
    const methods = Array.isArray(route.method) ? route.method : [route.method];
    for (const method of methods) {
      if (method === "HEAD" || method === "OPTIONS") continue;
      registeredRoutes.push({ method, url: route.url });
    }
  });

  const database = createDb(config.DATABASE_URL);
  app.decorate("database", database);
  app.setValidatorCompiler(validatorCompiler);
  app.setSerializerCompiler(serializerCompiler);
  await app.register(cors, { origin: false });
  await app.register(sensible);

  if (config.ENABLE_DOCS) {
    const [{ default: swagger }, { jsonSchemaTransform }, swaggerUi] = await Promise.all([
      import("@fastify/swagger"),
      import("fastify-type-provider-zod"),
      import("@fastify/swagger-ui"),
    ]);
    await app.register(swagger, {
      openapi: { info: { title: `${config.BRAND_NAME} API`, version: "1.0.0" } },
      transform: jsonSchemaTransform,
    });
    await app.register(swaggerUi.default, { routePrefix: "/docs" });
    app.get("/openapi.json", async () => app.swagger());
  }

  await app.register(requestId);
  await app.register(errorHandler);
  await app.register(rateLimit);
  await app.register(authPlugin);
  await app.register(healthRoutes);
  await app.register(authRoutes, { prefix: "/api/v1/auth" });
  await app.register(sessionRoutes, { prefix: "/api/v1/sessions" });
  await app.register(sendRoutes, { prefix: "/api/v1" });
  await app.register(messageRoutes, { prefix: "/api/v1/messages" });
  await app.register(webhookRoutes, { prefix: "/api/v1/sessions" });

  const purgeTimer = setInterval(() => {
    void purgeAuthArtifacts(database).catch(() => undefined);
  }, 60 * 60_000);
  purgeTimer.unref?.();

  app.addHook("onClose", async () => {
    clearInterval(purgeTimer);
    await closeQueues();
    await database.client.end();
  });
  return app;
}

export async function buildInternalApp(
  database: ReturnType<typeof createDb>,
): Promise<FastifyInstance> {
  const app = Fastify({
    logger: { level: config.LOG_LEVEL },
    trustProxy: false,
  });
  app.decorate("database", database);
  const adminRoutes = await import("./routes/internal/admin.js");
  await app.register(adminRoutes.default, { prefix: "/internal/admin" });
  return app;
}

import fp from "fastify-plugin";
import jwt from "@fastify/jwt";
import { and, eq, isNull } from "drizzle-orm";
import { accounts, personalAccessTokens, sessionApiKeys, users } from "@waapi/db";
import { config } from "../config.js";
import { sha256 } from "../lib/crypto.js";
import type { Identity } from "../services/auth.js";

export type AuthType = "jwt" | "pat" | "session_key";
export type RequestIdentity = Identity & {
  authType: AuthType;
  sessionId?: string;
  tokenHash?: string;
  scopes?: string[];
};

const httpError = (message: string, statusCode: number, code: string) =>
  Object.assign(new Error(message), { statusCode, code });

async function touchLastUsed(
  update: () => Promise<unknown>,
  lastUsedAt: Date | null,
): Promise<void> {
  if (lastUsedAt && Date.now() - lastUsedAt.getTime() < 60_000) return;
  await update();
}

export default fp(async (app) => {
  await app.register(jwt, {
    secret: config.JWT_SECRET,
    sign: {
      expiresIn: config.JWT_ACCESS_TTL,
      algorithm: "HS256",
      iss: config.JWT_ISSUER,
      aud: config.JWT_AUDIENCE,
    },
    verify: {
      algorithms: ["HS256"],
      allowedIss: config.JWT_ISSUER,
      allowedAud: config.JWT_AUDIENCE,
    },
  });

  app.decorateRequest("identity", null as unknown as RequestIdentity);

  app.decorate("authenticate", async (request, _reply) => {
    const header = request.headers.authorization;
    if (!header?.startsWith("Bearer ")) {
      throw httpError("Bearer token required", 401, "UNAUTHORIZED");
    }
    const token = header.slice(7);

    if (token.startsWith("waapi_pat_")) {
      const hash = sha256(token);
      const [row] = await app.database.db
        .select()
        .from(personalAccessTokens)
        .where(and(eq(personalAccessTokens.tokenHash, hash), isNull(personalAccessTokens.revokedAt)))
        .limit(1);
      if (!row) throw httpError("Invalid token", 401, "UNAUTHORIZED");
      const [account] = await app.database.db
        .select()
        .from(accounts)
        .where(eq(accounts.id, row.accountId))
        .limit(1);
      const [user] = await app.database.db
        .select()
        .from(users)
        .where(eq(users.id, row.userId))
        .limit(1);
      if (!account || account.suspendedAt || !user || user.disabledAt) {
        throw httpError("Forbidden", 403, "FORBIDDEN");
      }
      await touchLastUsed(
        () =>
          app.database.db
            .update(personalAccessTokens)
            .set({ lastUsedAt: new Date() })
            .where(eq(personalAccessTokens.id, row.id)),
        row.lastUsedAt,
      );
      request.identity = {
        userId: row.userId,
        accountId: row.accountId,
        role: "member",
        isAdmin: false,
        authType: "pat",
        tokenHash: hash,
        scopes: row.scopes ?? ["account"],
      };
      return;
    }

    if (token.startsWith("waapi_sk_")) {
      const hash = sha256(token);
      const [row] = await app.database.db
        .select()
        .from(sessionApiKeys)
        .where(and(eq(sessionApiKeys.tokenHash, hash), isNull(sessionApiKeys.revokedAt)))
        .limit(1);
      if (!row) throw httpError("Invalid token", 401, "UNAUTHORIZED");
      const [account] = await app.database.db
        .select()
        .from(accounts)
        .where(eq(accounts.id, row.accountId))
        .limit(1);
      if (!account || account.suspendedAt) throw httpError("Forbidden", 403, "FORBIDDEN");
      await touchLastUsed(
        () =>
          app.database.db
            .update(sessionApiKeys)
            .set({ lastUsedAt: new Date() })
            .where(eq(sessionApiKeys.id, row.id)),
        row.lastUsedAt,
      );
      request.identity = {
        userId: row.id,
        accountId: row.accountId,
        role: "session",
        isAdmin: false,
        authType: "session_key",
        sessionId: row.sessionId,
        tokenHash: hash,
      };
      return;
    }

    await request.jwtVerify();
    const payload = request.user;
    const [user] = await app.database.db
      .select()
      .from(users)
      .where(eq(users.id, payload.userId))
      .limit(1);
    const [account] = await app.database.db
      .select()
      .from(accounts)
      .where(eq(accounts.id, payload.accountId))
      .limit(1);
    if (!user || user.disabledAt || !account || account.suspendedAt) {
      throw httpError("Forbidden", 403, "FORBIDDEN");
    }
    request.identity = { ...payload, authType: "jwt", tokenHash: sha256(token) };
  });

  app.decorate("requireAuthType", (...types: AuthType[]) => {
    return async (
      request: import("fastify").FastifyRequest,
      _reply: import("fastify").FastifyReply,
    ) => {
      await app.authenticate(request, _reply);
      if (!types.includes(request.identity.authType)) {
        throw httpError("Forbidden for this credential type", 403, "FORBIDDEN");
      }
      if (request.identity.authType === "pat") {
        const scopes = request.identity.scopes ?? [];
        if (!scopes.includes("account") && !scopes.includes("*")) {
          throw httpError("PAT missing account scope", 403, "FORBIDDEN");
        }
      }
    };
  });
});

===== apps/api/src/plugins/rate-limit.ts =====
import fp from "fastify-plugin";
import rateLimit from "@fastify/rate-limit";
import { config } from "../config.js";
import { clientIp } from "../lib/client-ip.js";
import { sha256 } from "../lib/crypto.js";
import { redis } from "../services/queues.js";

export default fp(async (app) => {
  await app.register(rateLimit, {
    global: true,
    max: config.RATE_LIMIT_MAX,
    timeWindow: config.RATE_LIMIT_WINDOW,
    redis,
    keyGenerator: (request) => {
      const auth = request.headers.authorization;
      if (typeof auth === "string" && auth.startsWith("Bearer ")) {
        const token = auth.slice(7);
        if (request.identity?.accountId) {
          return `acct:${request.identity.accountId}`;
        }
        return `tok:${sha256(token).slice(0, 32)}`;
      }
      return `ip:${clientIp(request)}`;
    },
    errorResponseBuilder: (_request, context) => ({
      error: {
        code: "RATE_LIMITED",
        message: `Rate limit exceeded, retry after ${context.after}`,
        requestId: _request.id,
      },
    }),
  });
});

===== apps/api/src/lib/client-ip.ts =====
import type { FastifyRequest } from "fastify";

function trustCloudflare(): boolean {
  const raw = process.env.TRUST_CLOUDFLARE;
  if (raw == null) return false;
  return ["1", "true", "yes", "on"].includes(raw.toLowerCase());
}

/**
 * Single client-IP helper for lockout, rate limits, and audit.
 *
 * - When TRUST_CLOUDFLARE=true: prefer CF-Connecting-IP (only safe behind CF +
 *   nginx real_ip that overwrites the header; VPS firewall must allow 80/443
 *   from Cloudflare only â€” see deploy/nginx.conf).
 * - Otherwise: use request.ip from Fastify with trustProxy = TRUST_PROXY_HOPS
 *   (numeric hop count, default 1). Do not read X-Forwarded-For here.
 */
export function clientIp(request: FastifyRequest): string {
  if (trustCloudflare()) {
    const cf = request.headers["cf-connecting-ip"];
    if (typeof cf === "string" && cf.trim()) {
      return cf.split(",")[0]!.trim();
    }
  }
  return request.ip;
}

===== apps/api/src/lib/crypto.ts =====
import { createCipheriv, createDecipheriv, createHash, randomBytes, timingSafeEqual } from "node:crypto";
import { config } from "../config.js";

export const sha256 = (value: string): string =>
  createHash("sha256").update(value).digest("hex");

export const randomToken = (prefix = ""): string =>
  prefix + randomBytes(32).toString("base64url");

export function hashesEqual(a: string, b: string): boolean {
  const left = Buffer.from(a);
  const right = Buffer.from(b);
  return left.length === right.length && timingSafeEqual(left, right);
}

function key(): Buffer {
  const value = config.WEBHOOK_SECRET_KEY;
  const decoded = /^[0-9a-f]{64}$/i.test(value)
    ? Buffer.from(value, "hex")
    : Buffer.from(value, "base64");
  if (decoded.length !== 32) throw new Error("WEBHOOK_SECRET_KEY must decode to 32 bytes");
  return decoded;
}

export function encryptSecret(value: string): string {
  const iv = randomBytes(12);
  const cipher = createCipheriv("aes-256-gcm", key(), iv);
  const ciphertext = Buffer.concat([cipher.update(value, "utf8"), cipher.final()]);
  return [iv, cipher.getAuthTag(), ciphertext].map((part) => part.toString("base64url")).join(".");
}

export function decryptSecret(value: string): string {
  const [ivText, tagText, ciphertextText] = value.split(".");
  if (!ivText || !tagText || !ciphertextText) throw new Error("Invalid encrypted secret");
  const decipher = createDecipheriv("aes-256-gcm", key(), Buffer.from(ivText, "base64url"));
  decipher.setAuthTag(Buffer.from(tagText, "base64url"));
  return Buffer.concat([
    decipher.update(Buffer.from(ciphertextText, "base64url")),
    decipher.final(),
  ]).toString("utf8");
}

import { parsePhoneNumberFromString } from "libphonenumber-js";

const JID_RE = /^[1-9]\d{6,14}@s\.whatsapp\.net$/;

export function toWhatsAppJid(input: string): string {
  const trimmed = input.trim();
  if (JID_RE.test(trimmed)) return trimmed;

  const phone = parsePhoneNumberFromString(trimmed);
  if (!phone?.isValid()) {
    const error = new Error("Phone number must be valid E.164 or WhatsApp JID");
    Object.assign(error, { statusCode: 400, code: "INVALID_PHONE" });
    throw error;
  }
  return `${phone.number.slice(1)}@s.whatsapp.net`;
}

===== apps/api/src/services/auth.ts =====
import argon2 from "argon2";
import { randomUUID } from "node:crypto";
import { and, count, eq, gt, isNull, lt, or, sql } from "drizzle-orm";
import {
  accounts,
  emailVerificationTokens,
  loginAttempts,
  passwordResetTokens,
  refreshTokens,
  securityAuditLog,
  users,
  type Db,
} from "@waapi/db";
import { config } from "../config.js";
import { randomToken, sha256 } from "../lib/crypto.js";
import { defaultJobOptions, emailQueue } from "./queues.js";

const db = (value: Db) => value.db;

/** Precomputed argon2id hash so unknown-email logins still pay verify cost. */
let dummyHashPromise: Promise<string> | null = null;
const dummyHash = () => {
  dummyHashPromise ??= argon2.hash("timing-dummy-password-not-a-secret", {
    type: argon2.argon2id,
  });
  return dummyHashPromise;
};

const expiry = (ttl: string): Date => {
  const match = /^(\d+)([mhd])$/.exec(ttl);
  if (!match) throw new Error(`Invalid TTL: ${ttl}`);
  const unit = { m: 60_000, h: 3_600_000, d: 86_400_000 }[match[2]!]!;
  return new Date(Date.now() + Number(match[1]) * unit);
};

export type Identity = {
  userId: string;
  accountId: string;
  role: string;
  isAdmin: boolean;
};

async function audit(
  database: Db,
  identity: Partial<Identity>,
  eventType: string,
  ip?: string,
  meta?: Record<string, unknown>,
) {
  await db(database).insert(securityAuditLog).values({
    accountId: identity.accountId,
    userId: identity.userId,
    eventType,
    ip,
    meta: meta ?? {},
  });
}

async function issueRefresh(database: Db, userId: string, familyId = randomUUID()) {
  const token = randomToken();
  const [row] = await db(database)
    .insert(refreshTokens)
    .values({
      userId,
      familyId,
      tokenHash: sha256(token),
      expiresAt: expiry(config.REFRESH_TTL),
    })
    .returning();
  return { token, row: row! };
}

function isUniqueViolation(err: unknown): boolean {
  return (
    typeof err === "object" &&
    err !== null &&
    "code" in err &&
    (err as { code?: string }).code === "23505"
  );
}

export async function register(
  database: Db,
  input: { email: string; password: string; name?: string },
  ip?: string,
) {
  const email = input.email.trim().toLowerCase();
  const passwordHash = await argon2.hash(input.password, { type: argon2.argon2id });
  let result: { account: typeof accounts.$inferSelect; user: typeof users.$inferSelect; verificationToken: string };
  try {
    result = await db(database).transaction(async (tx) => {
      const [account] = await tx
        .insert(accounts)
        .values({ name: input.name ?? email })
        .returning();
      const [user] = await tx
        .insert(users)
        .values({ accountId: account!.id, email, passwordHash, role: "owner" })
        .returning();
      const token = randomToken();
      await tx.insert(emailVerificationTokens).values({
        userId: user!.id,
        tokenHash: sha256(token),
        expiresAt: expiry("24h"),
      });
      return { account: account!, user: user!, verificationToken: token };
    });
  } catch (err) {
    if (isUniqueViolation(err)) {
      throw Object.assign(new Error("Email already registered"), {
        statusCode: 409,
        code: "CONFLICT",
      });
    }
    throw err;
  }
  await emailQueue.add(
    "verify-email",
    {
      to: email,
      subject: "Verify your email",
      kind: "verify_email",
      // token only in job payload briefly; removeOnComplete clears it
      token: result.verificationToken,
    },
    defaultJobOptions,
  );
  await audit(database, { accountId: result.account.id, userId: result.user.id }, "auth.register", ip);
  return result;
}

export async function login(database: Db, emailInput: string, password: string, ip: string) {
  const email = emailInput.trim().toLowerCase();
  const since = new Date(Date.now() - 15 * 60_000);
  const [byEmail] = await db(database)
    .select({ value: count() })
    .from(loginAttempts)
    .where(
      and(
        eq(loginAttempts.email, email),
        eq(loginAttempts.succeeded, false),
        gt(loginAttempts.createdAt, since),
      ),
    );
  const [byIp] = await db(database)
    .select({ value: count() })
    .from(loginAttempts)
    .where(
      and(
        eq(loginAttempts.ip, ip),
        eq(loginAttempts.succeeded, false),
        gt(loginAttempts.createdAt, since),
      ),
    );
  if ((byEmail?.value ?? 0) >= 10 || (byIp?.value ?? 0) >= 30) {
    throw Object.assign(new Error("Too many login attempts"), {
      statusCode: 429,
      code: "BRUTE_FORCE_LOCKED",
    });
  }

  const [user] = await db(database).select().from(users).where(eq(users.email, email)).limit(1);
  const hash = user?.passwordHash ?? (await dummyHash());
  const valid = await argon2.verify(hash, password).catch(() => false);
  const ok = Boolean(user && valid);
  await db(database).insert(loginAttempts).values({ email, ip, succeeded: ok });
  if (!ok) {
    throw Object.assign(new Error("Invalid credentials"), {
      statusCode: 401,
      code: "INVALID_CREDENTIALS",
    });
  }
  if (user!.disabledAt) {
    throw Object.assign(new Error("Account disabled"), { statusCode: 403, code: "FORBIDDEN" });
  }
  if (!user!.emailVerifiedAt) {
    throw Object.assign(new Error("Email is not verified"), {
      statusCode: 403,
      code: "EMAIL_NOT_VERIFIED",
    });
  }
  const [account] = await db(database)
    .select()
    .from(accounts)
    .where(eq(accounts.id, user!.accountId))
    .limit(1);
  if (account?.suspendedAt) {
    throw Object.assign(new Error("Account suspended"), { statusCode: 403, code: "FORBIDDEN" });
  }
  const refresh = await issueRefresh(database, user!.id);
  await audit(database, { accountId: user!.accountId, userId: user!.id }, "auth.login", ip);
  return {
    identity: {
      userId: user!.id,
      accountId: user!.accountId,
      role: user!.role,
      isAdmin: user!.isAdmin,
    },
    refreshToken: refresh.token,
  };
}

export async function rotateRefresh(database: Db, token: string, ip?: string) {
  const hash = sha256(token);
  return db(database).transaction(async (tx) => {
    const claimed = await tx.execute(sql`
      update refresh_tokens
      set revoked_at = now()
      where token_hash = ${hash}
        and revoked_at is null
        and expires_at > now()
      returning id, user_id, family_id
    `);
    const current = claimed[0] as
      | { id: string; user_id: string; family_id: string }
      | undefined;

    if (!current) {
      const [existing] = await tx
        .select()
        .from(refreshTokens)
        .where(eq(refreshTokens.tokenHash, hash))
        .limit(1);
      if (existing) {
        await tx
          .update(refreshTokens)
          .set({ revokedAt: new Date() })
          .where(eq(refreshTokens.familyId, existing.familyId));
        await tx.insert(securityAuditLog).values({
          userId: existing.userId,
          eventType: "auth.refresh_reuse",
          ip,
          meta: { familyId: existing.familyId },
        });
      }
      throw Object.assign(new Error("Refresh token revoked"), {
        statusCode: 401,
        code: "TOKEN_REVOKED",
      });
    }

    const [user] = await tx.select().from(users).where(eq(users.id, current.user_id)).limit(1);
    if (!user || user.disabledAt) {
      throw Object.assign(new Error("User not found"), { statusCode: 401, code: "UNAUTHORIZED" });
    }
    const nextToken = randomToken();
    const [next] = await tx
      .insert(refreshTokens)
      .values({
        userId: user.id,
        familyId: current.family_id,
        tokenHash: sha256(nextToken),
        expiresAt: expiry(config.REFRESH_TTL),
      })
      .returning();
    await tx
      .update(refreshTokens)
      .set({ replacedBy: next!.id })
      .where(eq(refreshTokens.id, current.id));
    return {
      identity: {
        userId: user.id,
        accountId: user.accountId,
        role: user.role,
        isAdmin: user.isAdmin,
      },
      refreshToken: nextToken,
    };
  });
}

export async function logoutAll(database: Db, userId: string) {
  await db(database)
    .update(refreshTokens)
    .set({ revokedAt: new Date() })
    .where(and(eq(refreshTokens.userId, userId), isNull(refreshTokens.revokedAt)));
}

export async function verifyEmail(database: Db, token: string) {
  const hash = sha256(token);
  await db(database).transaction(async (tx) => {
    const claimed = await tx.execute(sql`
      update email_verification_tokens
      set used_at = now()
      where token_hash = ${hash}
        and used_at is null
        and expires_at > now()
      returning id, user_id
    `);
    const row = claimed[0] as { id: string; user_id: string } | undefined;
    if (!row) {
      throw Object.assign(new Error("Invalid verification token"), {
        statusCode: 400,
        code: "TOKEN_REVOKED",
      });
    }
    await tx
      .update(users)
      .set({ emailVerifiedAt: new Date(), updatedAt: new Date() })
      .where(eq(users.id, row.user_id));
  });
}

export async function resendVerification(database: Db, emailInput: string) {
  const email = emailInput.trim().toLowerCase();
  const [user] = await db(database).select().from(users).where(eq(users.email, email)).limit(1);
  if (!user || user.emailVerifiedAt) return;
  await db(database)
    .update(emailVerificationTokens)
    .set({ usedAt: new Date() })
    .where(
      and(
        eq(emailVerificationTokens.userId, user.id),
        isNull(emailVerificationTokens.usedAt),
      ),
    );
  const token = randomToken();
  await db(database).insert(emailVerificationTokens).values({
    userId: user.id,
    tokenHash: sha256(token),
    expiresAt: expiry("24h"),
  });
  await emailQueue.add(
    "verify-email",
    { to: email, kind: "verify_email", token },
    defaultJobOptions,
  );
}

export async function forgotPassword(database: Db, emailInput: string) {
  const email = emailInput.trim().toLowerCase();
  const [user] = await db(database).select().from(users).where(eq(users.email, email)).limit(1);
  if (!user) return;

  const since = new Date(Date.now() - 15 * 60_000);
  const [recent] = await db(database)
    .select({ value: count() })
    .from(passwordResetTokens)
    .where(
      and(eq(passwordResetTokens.userId, user.id), gt(passwordResetTokens.createdAt, since)),
    );
  if ((recent?.value ?? 0) >= 3) return;

  await db(database)
    .update(passwordResetTokens)
    .set({ usedAt: new Date() })
    .where(
      and(eq(passwordResetTokens.userId, user.id), isNull(passwordResetTokens.usedAt)),
    );

  const token = randomToken();
  await db(database).insert(passwordResetTokens).values({
    userId: user.id,
    tokenHash: sha256(token),
    expiresAt: expiry("1h"),
  });
  await emailQueue.add(
    "reset-password",
    { to: user.email, kind: "reset_password", token },
    defaultJobOptions,
  );
}

export async function resetPassword(database: Db, token: string, password: string) {
  await db(database).transaction(async (tx) => {
    const claimed = await tx.execute(sql`
      update password_reset_tokens
      set used_at = now()
      where token_hash = ${sha256(token)}
        and used_at is null
        and expires_at > now()
      returning id, user_id
    `);
    const row = claimed[0] as { id: string; user_id: string } | undefined;
    if (!row) {
      throw Object.assign(new Error("Invalid reset token"), {
        statusCode: 400,
        code: "TOKEN_REVOKED",
      });
    }
    await tx
      .update(users)
      .set({
        passwordHash: await argon2.hash(password, { type: argon2.argon2id }),
        updatedAt: new Date(),
      })
      .where(eq(users.id, row.user_id));
    await tx
      .update(refreshTokens)
      .set({ revokedAt: new Date() })
      .where(eq(refreshTokens.userId, row.user_id));
  });
}

  await db(database)

}

export async function createSession(
  database: Db,
  accountId: string,
  name: string,
  requestId: string,

  const session = await database.db.transaction(async (tx) => {
    await tx.execute(sql`select id from accounts where id = ${accountId} for update`);
    const accountRows = await tx.execute(
      sql`select id, max_sessions, suspended_at from accounts where id = ${accountId}`,
    );

      | { id: string; max_sessions: number; suspended_at: Date | null }
      | undefined;
    if (!account) {
      throw Object.assign(new Error("Account not found"), { statusCode: 404, code: "NOT_FOUND" });
    }
    if (account.suspended_at) {
      throw Object.assign(new Error("Account suspended"), { statusCode: 403, code: "FORBIDDEN" });
    }

    const [global] = await tx.select({ value: count() }).from(sessions).where(ne(sessions.status, "logged_out"));
    if ((global?.value ?? 0) >= config.GLOBAL_MAX_SESSIONS) {
      throw Object.assign(new Error("Global session capacity reached"), {
        statusCode: 503,
        code: "GLOBAL_SESSION_CAP",
      });
    }

    const [total] = await tx
      .select({ value: count() })
      .from(sessions)
      .where(and(eq(sessions.accountId, accountId), ne(sessions.status, "logged_out")));
    if ((total?.value ?? 0) >= Number(account.max_sessions)) {
      throw Object.assign(new Error("Account session cap reached"), {
        statusCode: 409,
        code: "ACCOUNT_SESSION_CAP",
      });
    }
    const [row] = await tx.insert(sessions).values({ accountId, name }).returning();
    return row!;
  });

  try {
    await sessionCommandsQueue.add(
      "open-session",
      {
        type: "open-session",
        accountId,
        sessionId: session.id,
        requestId,
      },
      { ...defaultJobOptions, jobId: `open-${session.id}-${Date.now()}` },
    );
  } catch (err) {
    await database.db.delete(sessions).where(eq(sessions.id, session.id));
    throw Object.assign(err instanceof Error ? err : new Error(String(err)), {
      statusCode: 503,
      code: "INTERNAL_ERROR",
    });
  }
  return session;
}

import { createHash, randomUUID } from "node:crypto";
import { and, eq, sql } from "drizzle-orm";
import { idempotencyKeys, outboundMessages, sessions, type Db } from "@waapi/db";
import { toWhatsAppJid } from "../lib/phone.js";
import { defaultJobOptions, sendTextQueue } from "./queues.js";

type SendInput = { sessionId: string; to: string; text: string };
type SendResponse = { id: string; status: "queued" | "failed" };

export async function sendText(
  database: Db,
  accountId: string,
  input: SendInput,
  key: string,
  requestId: string,
): Promise<SendResponse> {
  const bodyHash = createHash("sha256").update(JSON.stringify(input)).digest("hex");

  // Claim idempotency slot first so concurrent duplicates share one winner.
  await database.db.execute(sql`
    insert into idempotency_keys (id, account_id, key, request_hash, response_status, response_body, expires_at)
    values (
      ${randomUUID()}::uuid,
      ${accountId}::uuid,
      ${key},
      ${bodyHash},
      202,
      ${JSON.stringify({ id: null, status: "queued" })}::jsonb,
      ${new Date(Date.now() + 86_400_000).toISOString()}::timestamptz
    )
    on conflict (account_id, key) do nothing
  `);

  const [prior] = await database.db
    .select()
    .from(idempotencyKeys)
    .where(and(eq(idempotencyKeys.accountId, accountId), eq(idempotencyKeys.key, key)))
    .limit(1);
  if (!prior) {
    throw Object.assign(new Error("Idempotency claim failed"), {
      statusCode: 500,
      code: "INTERNAL_ERROR",
    });
  }
  if (prior.requestHash !== bodyHash) {
    throw Object.assign(new Error("Idempotency key reused with different body"), {
      statusCode: 409,
      code: "IDEMPOTENCY_CONFLICT",
    });
  }
  const priorBody = prior.responseBody as { id?: string | null; status?: string };
  if (priorBody.id) {
    return priorBody as SendResponse;
  }

  // Winner continues: create outbound row then enqueue after commit.
  const [session] = await database.db
    .select()
    .from(sessions)
    .where(and(eq(sessions.id, input.sessionId), eq(sessions.accountId, accountId)))
    .limit(1);
  if (!session) {
    throw Object.assign(new Error("Session not found"), { statusCode: 404, code: "NOT_FOUND" });
  }
  if (session.status !== "connected") {
    throw Object.assign(new Error("Session is not connected"), {
      statusCode: 409,
      code: "SESSION_NOT_CONNECTED",
    });
  }

  const to = toWhatsAppJid(input.to);
  const message = await database.db.transaction(async (tx) => {
    // Re-check if another winner already filled the response.
    const [again] = await tx
      .select()
      .from(idempotencyKeys)
      .where(and(eq(idempotencyKeys.accountId, accountId), eq(idempotencyKeys.key, key)))
      .limit(1);
    const againBody = again?.responseBody as { id?: string | null } | undefined;
    if (againBody?.id) {
      return { id: againBody.id, status: "queued" as const, skipEnqueue: true };
    }

    const [row] = await tx
      .insert(outboundMessages)
      .values({ accountId, sessionId: input.sessionId, toJid: to, idempotencyKey: key })
      .returning();
    const response: SendResponse = { id: row!.id, status: "queued" };
    await tx
      .update(idempotencyKeys)
      .set({ responseStatus: 202, responseBody: response })
      .where(eq(idempotencyKeys.id, prior.id));
    return { ...response, skipEnqueue: false };
  });

  if (message.skipEnqueue) {
    return { id: message.id, status: "queued" };
  }

  try {
    await sendTextQueue.add(
      "send-text",
      {
        accountId,
        sessionId: input.sessionId,
        to,
        text: input.text,
        messageId: message.id,
        idempotencyKey: key,
        requestId,
      },
      {
        ...defaultJobOptions,
        jobId: message.id,
        attempts: 5,
        backoff: { type: "exponential", delay: 1_000 },
      },
    );
  } catch (err) {
    await database.db
      .update(outboundMessages)
      .set({
        status: "failed",
        error: err instanceof Error ? err.message : String(err),
        updatedAt: new Date(),
      })
      .where(eq(outboundMessages.id, message.id));
    throw Object.assign(err instanceof Error ? err : new Error(String(err)), {
      statusCode: 503,
      code: "INTERNAL_ERROR",
    });
  }

  return { id: message.id, status: "queued" };
}

import { lookup as dnsLookup } from "node:dns/promises";
import { isIP } from "node:net";
import { Agent, type Dispatcher } from "undici";
import ipaddr from "ipaddr.js";

function allowHttpWebhooks(): boolean {
  const raw = process.env.ALLOW_HTTP_WEBHOOKS;
  return raw != null && ["1", "true", "yes", "on"].includes(raw.toLowerCase());
}

export type LookupFn = (
  hostname: string,
  options: unknown,
  callback: (err: NodeJS.ErrnoException | null, address: string, family: number) => void,
) => void;

function normalizeHostname(hostname: string): string {
  if (hostname.startsWith("[") && hostname.endsWith("]")) {
    return hostname.slice(1, -1);
  }
  return hostname;
}

/** Expand decimal / dotted-hex / octal-ish IPv4 literals into dotted decimal when possible. */
function coerceIpLiteral(hostname: string): string | null {
  const host = normalizeHostname(hostname);
  if (isIP(host)) return host;

  // Pure decimal IPv4 (e.g. 2130706433)
  if (/^\d+$/.test(host)) {
    const n = Number(host);
    if (n >= 0 && n <= 0xffffffff) {
      return [(n >>> 24) & 255, (n >>> 16) & 255, (n >>> 8) & 255, n & 255].join(".");
    }
  }

  // Dotted hex/octal forms like 0x7f.1 or 0177.0.0.1 â€” reject by attempting ipaddr parse after
  // converting each octet if hex/octal prefixed.
  if (host.includes(".")) {
    const parts = host.split(".");
    if (parts.length === 4 || parts.length === 2) {
      try {
        const nums = parts.map((part) => {
          if (/^0x[0-9a-f]+$/i.test(part)) return Number.parseInt(part, 16);
          if (/^0[0-7]+$/.test(part)) return Number.parseInt(part, 8);
          if (/^\d+$/.test(part)) return Number(part);
          throw new Error("bad");
        });
        if (nums.length === 2) {
          // 0x7f.1 â†’ 127.0.0.1 style (a.b as 127.1)
          const a = nums[0]!;
          const b = nums[1]!;
          if (a <= 255 && b <= 0xffffff) {
            return [a, (b >>> 16) & 255, (b >>> 8) & 255, b & 255].join(".");
          }
        }
        if (nums.every((n) => n >= 0 && n <= 255)) return nums.join(".");
      } catch {
        /* fall through */
      }
    }
  }
  return null;
}

export function isBlockedAddress(address: string): boolean {
  let parsed: ipaddr.IPv4 | ipaddr.IPv6;
  try {
    parsed = ipaddr.parse(address);
  } catch {
    return true;
  }

  if (parsed.kind() === "ipv6") {
    const v6 = parsed as ipaddr.IPv6;
    if (v6.isIPv4MappedAddress()) {
      return isBlockedAddress(v6.toIPv4Address().toString());
    }
    const range = v6.range();
    if (
      range === "unspecified" ||
      range === "loopback" ||
      range === "uniqueLocal" ||
      range === "linkLocal" ||
      range === "multicast" ||
      range === "reserved"
    ) {
      return true;
    }
    // fc00::/7 unique local already covered; ff00::/8 multicast covered
    return false;
  }

  const v4 = parsed as ipaddr.IPv4;
  const range = v4.range();
  if (
    range === "unspecified" ||
    range === "broadcast" ||
    range === "multicast" ||
    range === "linkLocal" ||
    range === "loopback" ||
    range === "private" ||
    range === "reserved" ||
    range === "carrierGradeNat" ||
    range === "benchmarking"
  ) {
    return true;
  }
  // 192.0.0.0/24 (IETF protocol assignments) â€” ipaddr may classify as reserved; double-check
  const octets = v4.octets;
  if (octets[0] === 192 && octets[1] === 0 && octets[2] === 0) return true;
  if (octets[0]! >= 240) return true;
  return false;
}

export async function resolveAndValidate(
  hostname: string,
  lookupImpl: typeof dnsLookup = dnsLookup,
): Promise<{ address: string; family: number }> {
  const host = normalizeHostname(hostname);
  const literal = coerceIpLiteral(host);
  if (literal) {
    if (isBlockedAddress(literal)) {
      throw Object.assign(new Error("Webhook URL resolves to a private address"), {
        statusCode: 400,
        code: "SSRF_BLOCKED",
      });
    }
    return { address: literal, family: isIP(literal) === 6 ? 6 : 4 };
  }

  const results = await lookupImpl(host, { all: true, verbatim: true });
  const list = Array.isArray(results) ? results : [results];
  if (!list.length) {
    throw Object.assign(new Error("Webhook URL DNS lookup failed"), {
      statusCode: 400,
      code: "SSRF_BLOCKED",
    });
  }
  for (const entry of list) {
    if (isBlockedAddress(entry.address)) {
      throw Object.assign(new Error("Webhook URL resolves to a private address"), {
        statusCode: 400,
        code: "SSRF_BLOCKED",
      });
    }
  }
  const first = list[0]!;
  return { address: first.address, family: first.family };
}

export function assertWebhookUrlShape(value: string): URL {
  let url: URL;
  try {
    url = new URL(value);
  } catch {
    throw Object.assign(new Error("Invalid webhook URL"), {
      statusCode: 400,
      code: "SSRF_BLOCKED",
    });
  }
  if (url.username || url.password) {
    throw Object.assign(new Error("Webhook URL must not include userinfo"), {
      statusCode: 400,
      code: "SSRF_BLOCKED",
    });
  }
  if (url.protocol === "https:") return url;
  if (url.protocol === "http:" && allowHttpWebhooks()) return url;
  throw Object.assign(new Error("Webhook URL must use HTTPS"), {
    statusCode: 400,
    code: "SSRF_BLOCKED",
  });
}

/** Legacy name used by upsert path â€” shape + resolve check only (connect pins IP). */
export async function assertPublicUrl(value: string): Promise<URL> {
  const url = assertWebhookUrlShape(value);
  await resolveAndValidate(url.hostname);
  return url;
}

export function createPinnedAgent(options?: {
  lookup?: LookupFn;
  connectTimeout?: number;
}): Agent {
  return new Agent({
    connect: {
      timeout: options?.connectTimeout ?? 10_000,
      lookup: (hostname, opts, cb) => {
        const run = async () => {
          if (options?.lookup) {
            await new Promise<void>((resolve, reject) => {
              options.lookup!(hostname, opts, (err, address, family) => {
                if (err) reject(err);
                else {
                  cb(null, address, family);
                  resolve();
                }
              });
            });
            return;
          }
          const validated = await resolveAndValidate(hostname);
          cb(null, validated.address, validated.family);
        };
        void run().catch((err: NodeJS.ErrnoException) => cb(err, "", 4));
      },
    },
  });
}

export const MAX_WEBHOOK_RESPONSE_BYTES = 64 * 1024;

export async function dumpBodyLimited(
  body: Dispatcher.ResponseData["body"],
  maxBytes = MAX_WEBHOOK_RESPONSE_BYTES,
): Promise<void> {
  let total = 0;
  for await (const chunk of body) {
    total += chunk.length;
    if (total > maxBytes) {
      body.destroy();
      throw new Error("Webhook response exceeded 64KB");
    }
  }
}

import { createHmac, timingSafeEqual } from "node:crypto";

export function signPayload(secret: string, timestamp: number, body: string): string {
  return createHmac("sha256", secret).update(`${timestamp}.${body}`).digest("hex");
}

export const signWebhook = signPayload;

export function verifySignature(
  secret: string,
  timestamp: number,
  body: string,
  signature: string,
): boolean {
  const supplied = signature.startsWith("sha256=") ? signature.slice(7) : signature;
  const expected = signPayload(secret, timestamp, body);
  if (!/^[0-9a-f]{64}$/i.test(supplied)) return false;
  return timingSafeEqual(Buffer.from(expected, "hex"), Buffer.from(supplied, "hex"));
}

import { eq } from "drizzle-orm";

import { QUEUE_WEBHOOK_DELIVERY } from "@waapi/shared";
import { webhookDeliveries, webhookEndpoints, type Db } from "@waapi/db";

import { redis } from "../services/queues.js";
import { signWebhook } from "../services/webhooks/sign.js";

  assertWebhookUrlShape,
  createPinnedAgent,
  dumpBodyLimited,

function backoffWithJitter(attempts: number): number {
  const base = Math.min(3_600_000, 1_000 * 2 ** attempts);
  const jitter = Math.floor(Math.random() * Math.min(1_000, base * 0.2));
  return base + jitter;
}

export function startWebhookDeliveryWorker(database: Db): Worker {
  const agent = createPinnedAgent();

  return new Worker(
    QUEUE_WEBHOOK_DELIVERY,
    async (job) => {
      const [delivery] = await database.db
        .select()
        .from(webhookDeliveries)
        .where(eq(webhookDeliveries.id, job.data.deliveryId as string))
        .limit(1);
      if (!delivery || delivery.status === "success" || delivery.status === "dead" || delivery.status === "skipped") {
        return;
      }
      const [endpoint] = await database.db
        .select()
        .from(webhookEndpoints)
        .where(eq(webhookEndpoints.id, delivery.endpointId))
        .limit(1);

      try {
        if (!endpoint) {
          await database.db
            .update(webhookDeliveries)
            .set({
              status: "skipped",
              lastError: "endpoint_missing",
              attempts: delivery.attempts + 1,
              updatedAt: new Date(),
            })
            .where(eq(webhookDeliveries.id, delivery.id));
          return;
        }
        if (!endpoint.enabled) {
          await database.db
            .update(webhookDeliveries)
            .set({
              status: "skipped",
              lastError: "endpoint_disabled",
              attempts: delivery.attempts + 1,
              updatedAt: new Date(),
            })
            .where(eq(webhookDeliveries.id, delivery.id));
          return;
        }

        const url = assertWebhookUrlShape(endpoint.url);
        const body = JSON.stringify(delivery.payload);
        const timestamp = Math.floor(Date.now() / 1000);
        const response = await request(url.toString(), {
          method: "POST",
          headers: {
            "content-type": "application/json",
            "user-agent": "waapi-webhooks/1.0",
            "x-waapi-event": delivery.eventType,
            "x-waapi-timestamp": String(timestamp),
            "x-waapi-delivery-id": delivery.id,
            "x-waapi-signature": `sha256=${signWebhook(decryptSecret(endpoint.secretEncrypted), timestamp, body)}`,
          },
          body,
          dispatcher: agent,
          headersTimeout: 10_000,
          bodyTimeout: 10_000,
          // undici request does not follow redirects by default (maxRedirections: 0)
        } as Parameters<typeof request>[1]);
        await dumpBodyLimited(response.body);
        if (response.statusCode < 200 || response.statusCode >= 300) {
          throw new Error(`HTTP ${response.statusCode}`);
        }
        await database.db
          .update(webhookDeliveries)
          .set({
            status: "success",
            attempts: delivery.attempts + 1,
            responseCode: response.statusCode,
            updatedAt: new Date(),
          })
          .where(eq(webhookDeliveries.id, delivery.id));
      } catch (error) {
        const final = job.attemptsMade + 1 >= 8;
        await database.db
          .update(webhookDeliveries)
          .set({
            status: final ? "dead" : "retrying",
            attempts: delivery.attempts + 1,
            lastError: error instanceof Error ? error.message : String(error),
            updatedAt: new Date(),
          })
          .where(eq(webhookDeliveries.id, delivery.id));
        throw error;
      }
    },
    {
      connection: redis,
      concurrency: 10,
      settings: { backoffStrategy: backoffWithJitter },
    },
  );
}

import { Worker } from "bullmq";

async function withNotFoundRetry<T>(fn: () => Promise<T>, attempts = 5): Promise<T> {
  let last: unknown;
  for (let i = 0; i < attempts; i++) {
    try {
      return await fn();
    } catch (err) {
      last = err;
      const message = err instanceof Error ? err.message : String(err);
      if (message !== "SESSION_NOT_FOUND_FOR_ACCOUNT" || i === attempts - 1) throw err;
      await new Promise((r) => setTimeout(r, 50 * (i + 1)));
    }
  }
  throw last;
}

export function startSessionCommandsWorker(args: {
  connection: Redis;
  manager: SessionManager;
  log: Logger;
}): Worker {
  const { connection, manager, log } = args;

  return new Worker(
    QUEUE_SESSION_COMMANDS,
    async (job) => {
      const cmd = workerCommandSchema.parse(job.data);
      log.info({ type: cmd.type, sessionId: cmd.sessionId }, "session command");
      switch (cmd.type) {
        case "open-session":
          await withNotFoundRetry(() => manager.openSession(cmd.accountId, cmd.sessionId));
          break;
        case "close-session":
          await withNotFoundRetry(() => manager.closeSession(cmd.accountId, cmd.sessionId));
          break;
        case "logout-session":
          await withNotFoundRetry(() => manager.logoutSession(cmd.accountId, cmd.sessionId));
          break;
        case "send-message":
          throw new Error("SEND_MESSAGE_WRONG_QUEUE");
        default:
          break;
      }
    },
    {
      connection,
      concurrency: 5,
      removeOnComplete: { age: 60, count: 200 },
      removeOnFail: { age: 300, count: 200 },
    },
  );
}

===== apps/worker/src/jobs/send-text.ts =====
import { Worker } from "bullmq";
import { and, eq, sql } from "drizzle-orm";
import { outboundMessages } from "@waapi/db";
import { QUEUE_SEND_TEXT, sendTextJobSchema } from "@waapi/shared";
import type { Redis } from "ioredis";
import type { Db } from "../db/client.js";
import { insertMessageLog } from "../db/message-logs.js";
import { getSessionForAccount } from "../db/sessions.js";
import { enqueueWebhookOutbox } from "../db/webhook-outbox.js";
import type { Logger } from "../logger.js";
import type { SessionManager } from "../session/manager.js";

const lastSendAt = new Map<string, number>();
const SEND_MIN_INTERVAL_MS = Number(process.env.SEND_MIN_INTERVAL_MS ?? 800);
const BODY_PLACEHOLDER = "[redacted]";

async function throttleSession(sessionId: string): Promise<void> {
  const last = lastSendAt.get(sessionId) ?? 0;
  const jitter = Math.floor(Math.random() * 400);
  const wait = SEND_MIN_INTERVAL_MS + jitter - (Date.now() - last);
  if (wait > 0) await new Promise((r) => setTimeout(r, wait));
  lastSendAt.set(sessionId, Date.now());
}

async function withNotFoundRetry<T>(fn: () => Promise<T>, attempts = 5): Promise<T> {
  let last: unknown;
  for (let i = 0; i < attempts; i++) {
    try {
      return await fn();
    } catch (err) {
      last = err;
      const message = err instanceof Error ? err.message : String(err);
      if (message !== "SESSION_NOT_FOUND_FOR_ACCOUNT" || i === attempts - 1) throw err;
      await new Promise((r) => setTimeout(r, 50 * (i + 1)));
    }
  }
  throw last;
}

/** Atomically claim a queued outbound row for sending. Exported for unit tests. */
export async function claimOutboundForSend(
  db: Db["db"],
  messageId: string,
): Promise<{ status: "claimed" | "already_sent" | "busy" | "missing" | "failed" }> {
  const claimed = await db.execute(sql`
    update outbound_messages
    set status = 'sending', updated_at = now()
    where id = ${messageId}::uuid and status = 'queued'
    returning id
  `);
  if (claimed[0]) return { status: "claimed" };

  const [row] = await db
    .select({
      status: outboundMessages.status,
      providerMessageId: outboundMessages.providerMessageId,
    })
    .from(outboundMessages)
    .where(eq(outboundMessages.id, messageId))
    .limit(1);
  if (!row) return { status: "missing" };
  if (row.status === "sent" || row.providerMessageId) return { status: "already_sent" };
  if (row.status === "failed") return { status: "failed" };
  return { status: "busy" };
}

async function redactJobBody(
  job: { updateData: (data: Record<string, unknown>) => Promise<void>; data: unknown },
  payload: Record<string, unknown>,
): Promise<void> {
  await job.updateData({ ...payload, text: BODY_PLACEHOLDER });
}

export function startSendTextWorker(args: {
  connection: Redis;
  db: Db["db"];
  manager: SessionManager;
  log: Logger;
}): Worker {
  const { connection, db, manager, log } = args;

  return new Worker(
    QUEUE_SEND_TEXT,
    async (job) => {
      const payload = sendTextJobSchema.parse(job.data);
      const text = payload.text;
      if (!text || text === BODY_PLACEHOLDER) {
        throw new Error("SEND_BODY_MISSING");
      }
      if (!payload.messageId) throw new Error("SEND_MESSAGE_ID_REQUIRED");

      const claim = await claimOutboundForSend(db, payload.messageId);
      if (claim.status === "already_sent") {
        await redactJobBody(job, payload);
        return { skipped: true, reason: "already_sent" };
      }
      if (claim.status === "busy") throw new Error("SEND_IN_PROGRESS");
      if (claim.status === "missing" || claim.status === "failed") {
        throw new Error(`SEND_CLAIM_${claim.status.toUpperCase()}`);
      }

      const isFinalAttempt = job.attemptsMade + 1 >= (job.opts.attempts ?? 1);

      try {
        const session = await withNotFoundRetry(() =>
          getSessionForAccount(db, payload.accountId, payload.sessionId).then((row) => {
            if (!row) throw new Error("SESSION_NOT_FOUND_FOR_ACCOUNT");
            return row;
          }),
        );
        if (session.status !== "connected") throw new Error("SESSION_NOT_CONNECTED");
        const client = manager.getClient(payload.sessionId);
        if (!client) throw new Error("SESSION_NOT_ON_NODE");

        await throttleSession(payload.sessionId);
        const result = await client.sendText(payload.to, text);

        await db
          .update(outboundMessages)
          .set({
            status: "sent",
            providerMessageId: result.messageId,
            updatedAt: new Date(),
          })
          .where(
            and(eq(outboundMessages.id, payload.messageId), eq(outboundMessages.status, "sending")),
          );

        await redactJobBody(job, payload);

        await insertMessageLog(db, {
          accountId: payload.accountId,
          sessionId: payload.sessionId,
          direction: "outbound",
          remoteJid: result.remoteJid,
          messageId: result.messageId,
          fromMe: true,
          messageType: "conversation",
        });

        await enqueueWebhookOutbox(db, {
          accountId: payload.accountId,
          sessionId: payload.sessionId,
          eventType: "message.sent",
          payload: {
            id: crypto.randomUUID(),
            type: "message.sent",
            accountId: payload.accountId,
            sessionId: payload.sessionId,
            createdAt: new Date().toISOString(),
            data: {
              messageId: result.messageId,
              remoteJid: result.remoteJid,
              outboundId: payload.messageId,
            },
          },
        });

        log.info(
          {
            sessionId: payload.sessionId,
            messageId: result.messageId,
            remoteJid: result.remoteJid,
            event: "message.sent",
          },
          "outbound message metadata",
        );
        return { messageId: result.messageId, remoteJid: result.remoteJid };
      } catch (err) {
        // Re-queue for retry unless this was the final attempt.
        if (isFinalAttempt) {
          await db
            .update(outboundMessages)
            .set({
              status: "failed",
              error: err instanceof Error ? err.message : String(err),
              updatedAt: new Date(),
            })
            .where(eq(outboundMessages.id, payload.messageId));
          await redactJobBody(job, payload);
        } else {
          await db
            .update(outboundMessages)
            .set({ status: "queued", updatedAt: new Date() })
            .where(
              and(
                eq(outboundMessages.id, payload.messageId),
                eq(outboundMessages.status, "sending"),
              ),
            );
        }
        throw err;
      }
    },
    {
      connection,
      concurrency: 5,
      removeOnComplete: { age: 60, count: 200 },
      // Body is redacted before final failure; keep failed jobs briefly only.
      removeOnFail: { age: 1, count: 50 },
    },
  );
}

import "dotenv/config";
import { z } from "zod";

const bool = (fallback: boolean) =>
  z
    .union([z.boolean(), z.string()])
    .optional()
    .transform((value) => {
      if (typeof value === "boolean") return value;
      if (value == null) return fallback;
      return ["1", "true", "yes", "on"].includes(value.toLowerCase());
    });

/** Values that must never be used in any environment (copied from docs/examples). */
const PLACEHOLDER_SECRETS = new Set([
  "",
  "change-me",
  "changeme",
  "secret",
  "password",
  "ci-jwt-secret-at-least-32-characters-long",
  "ci-admin-secret",
  "test-jwt-secret-that-is-at-least-32-bytes",
  "test-admin-api-secret",
  "AAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAA=",
]);

function decodeKey32(value: string, name: string): Buffer {
  const decoded = /^[0-9a-f]{64}$/i.test(value)
    ? Buffer.from(value, "hex")
    : Buffer.from(value, "base64");
  if (decoded.length !== 32) {
    throw new Error(`${name} must decode to exactly 32 bytes`);
  }
  return decoded;
}

function rejectPlaceholder(name: string, value: string): string {
  const trimmed = value.trim();
  if (PLACEHOLDER_SECRETS.has(trimmed) || PLACEHOLDER_SECRETS.has(trimmed.toLowerCase())) {
    throw new Error(`${name} is a placeholder or empty; set a real secret`);
  }
  if (/^A{16,}=*$/i.test(trimmed) || /^0{32,}$/.test(trimmed)) {
    throw new Error(`${name} looks like a placeholder; set a real secret`);
  }
  return trimmed;
}

const envSchema = z.object({
  NODE_ENV: z.enum(["development", "test", "production"]).default("development"),
  LOG_LEVEL: z.string().default("info"),
  API_PORT: z.coerce.number().int().min(1).max(65535).default(3000),
  INTERNAL_PORT: z.coerce.number().int().min(1).max(65535).default(3001),
  DATABASE_URL: z.string().min(1),
  REDIS_URL: z.string().min(1),
  JWT_SECRET: z.string().min(32),
  JWT_ACCESS_TTL: z.string().default("15m"),
  JWT_ISSUER: z.string().default("waapi"),

  RATE_LIMIT_REGISTER: z.coerce.number().int().positive().default(5),

  DEV_RETURN_TOKENS: bool(false),
  ENABLE_DOCS: bool(false),
  ALLOW_DOCS_IN_PROD: bool(false),
  ALLOW_HTTP_WEBHOOKS: bool(false),
  TRUST_PROXY_HOPS: z.coerce.number().int().min(0).max(5).default(1),
  TRUST_CLOUDFLARE: bool(false),
});

export type Config = z.infer<typeof envSchema>;

function loadConfig(env: NodeJS.ProcessEnv = process.env): Config {
  const parsed = envSchema.parse(env);
  // Always require a real 32-byte webhook encryption key.
  decodeKey32(parsed.WEBHOOK_SECRET_KEY, "WEBHOOK_SECRET_KEY");

  if (parsed.NODE_ENV === "production") {
    rejectPlaceholder("JWT_SECRET", parsed.JWT_SECRET);
    rejectPlaceholder("ADMIN_API_SECRET", parsed.ADMIN_API_SECRET);
    rejectPlaceholder("WEBHOOK_SECRET_KEY", parsed.WEBHOOK_SECRET_KEY);
    if (parsed.DEV_RETURN_TOKENS) {
      throw new Error("DEV_RETURN_TOKENS must be false in production");
    }
    if (parsed.ALLOW_HTTP_WEBHOOKS) {
      throw new Error("ALLOW_HTTP_WEBHOOKS must be false in production");
    }
    if (parsed.ENABLE_DOCS && !parsed.ALLOW_DOCS_IN_PROD) {
      throw new Error("ENABLE_DOCS requires ALLOW_DOCS_IN_PROD=true in production");
    }
  }

  return parsed;
}

    raw == null
      ? config.DEV_RETURN_TOKENS
      : ["1", "true", "yes", "on"].includes(raw.toLowerCase());
  return env !== "production" && flag;
};

import type { FastifyPluginAsync } from "fastify";
import {
  createPatBodySchema,
  forgotPasswordBodySchema,
  loginBodySchema,
  refreshBodySchema,
  registerBodySchema,
  resetPasswordBodySchema,
  verifyEmailBodySchema,
} from "@waapi/shared";
import { z } from "zod";
import { isDevTokenReturnEnabled } from "../../config.js";
import { authRateLimitConfig } from "../../lib/auth-rate-limit.js";
import { clientIp } from "../../lib/client-ip.js";
import { createPat, listPats, revokePat } from "../../services/tokens.js";
import {
  forgotPassword,
  login,
  logoutAll,
  register,
  resendVerification,
  resetPassword,
  rotateRefresh,
  verifyEmail,
} from "../../services/auth.js";

const routes: FastifyPluginAsync = async (app) => {
  const tokens = (identity: Parameters<typeof app.jwt.sign>[0], refreshToken: string) => ({
    accessToken: app.jwt.sign(identity),
    refreshToken,
  });

  app.post("/register", authRateLimitConfig("register"), async (request, reply) => {
    const result = await register(
      app.database,
      registerBodySchema.parse(request.body),
      clientIp(request),
    );
    const body: {
      userId: string;
      accountId: string;
      verificationToken?: string;
    } = {
      userId: result.user.id,
      accountId: result.account.id,
    };
    if (isDevTokenReturnEnabled()) {
      body.verificationToken = result.verificationToken;
    }
    return reply.code(201).send(body);
  });

  app.post("/login", authRateLimitConfig("login"), async (request) => {
    const input = loginBodySchema.parse(request.body);
    const result = await login(app.database, input.email, input.password, clientIp(request));
    return tokens(result.identity, result.refreshToken);
  });

  app.post("/refresh", authRateLimitConfig("refresh"), async (request) => {
    const result = await rotateRefresh(
      app.database,
      refreshBodySchema.parse(request.body).refreshToken,
      clientIp(request),
    );
    return tokens(result.identity, result.refreshToken);
  });

  app.post("/verify-email", authRateLimitConfig("verify-email"), async (request, reply) => {
    await verifyEmail(app.database, verifyEmailBodySchema.parse(request.body).token);
    return reply.code(204).send();
  });

  app.post(
    "/resend-verification",
    authRateLimitConfig("resend-verification"),
    async (request, reply) => {
      const email = z.object({ email: z.string().email().max(320) }).parse(request.body).email;
      await resendVerification(app.database, email);
      return reply.code(202).send();
    },
  );

  app.post("/forgot-password", authRateLimitConfig("forgot-password"), async (request, reply) => {
    await forgotPassword(app.database, forgotPasswordBodySchema.parse(request.body).email);
    return reply.code(202).send();
  });

  app.post("/reset-password", authRateLimitConfig("reset-password"), async (request, reply) => {
    const input = resetPasswordBodySchema.parse(request.body);
    await resetPassword(app.database, input.token, input.password);
    return reply.code(204).send();
  });

  app.post(
    "/logout-all",
    { preHandler: app.requireAuthType("jwt") },
    async (request, reply) => {
      await logoutAll(app.database, request.identity.userId);
      return reply.code(204).send();
    },
  );

  app.post(
    "/tokens",
    { preHandler: app.requireAuthType("jwt") },
    async (request, reply) => {
      const input = createPatBodySchema.parse(request.body);
      return reply
        .code(201)
        .send(
          await createPat(
            app.database,
            request.identity.userId,
            request.identity.accountId,
            input.name,
          ),
        );
    },
  );

  app.get("/tokens", { preHandler: app.requireAuthType("jwt") }, async (request) =>
    listPats(app.database, request.identity.accountId),
  );

  app.delete<{ Params: { id: string } }>(
    "/tokens/:id",
    { preHandler: app.requireAuthType("jwt") },
    async (request, reply) => {
      await revokePat(app.database, request.identity.accountId, request.params.id);
      return reply.code(204).send();
    },
  );
};
export default routes;

===== apps/api/src/lib/auth-rate-limit.ts =====
import type { FastifyRequest } from "fastify";
import { config } from "../config.js";
import { clientIp } from "./client-ip.js";

export type AuthRateBucket =
  | "login"
  | "register"
  | "refresh"
  | "forgot-password"
  | "reset-password"
  | "verify-email"
  | "resend-verification";

const LIMITS: Record<AuthRateBucket, () => number> = {
  login: () => config.RATE_LIMIT_LOGIN,
  register: () => config.RATE_LIMIT_REGISTER,
  refresh: () => config.RATE_LIMIT_REFRESH,
  "forgot-password": () => config.RATE_LIMIT_FORGOT_PASSWORD,
  "reset-password": () => config.RATE_LIMIT_RESET_PASSWORD,
  "verify-email": () => config.RATE_LIMIT_VERIFY_EMAIL,
  "resend-verification": () => config.RATE_LIMIT_RESEND_VERIFICATION,
};

/** Distinct Redis key per auth route so exhausting one bucket cannot starve another. */
export function authRateLimitKey(bucket: AuthRateBucket, request: FastifyRequest): string {
  return `auth:${bucket}:ip:${clientIp(request)}`;
}

export function authRateLimitConfig(bucket: AuthRateBucket) {
  return {
    config: {
      rateLimit: {
        max: LIMITS[bucket](),
        keyGenerator: (request: FastifyRequest) => authRateLimitKey(bucket, request),
      },
    },
  };
}

COPY packages/db/package.json packages/db/
RUN pnpm install --frozen-lockfile --filter @waapi/api...

FROM deps AS build
COPY . .
RUN pnpm --filter @waapi/shared build \
 && pnpm --filter @waapi/db build \
 && pnpm --filter @waapi/api build

FROM node:20-bookworm-slim AS runner
RUN corepack enable && corepack prepare pnpm@9.15.0 --activate \
 && useradd --system --uid 10001 --create-home waapi
WORKDIR /app
ENV NODE_ENV=production
ENV ENABLE_DOCS=false
ENV DEV_RETURN_TOKENS=false
ENV ALLOW_HTTP_WEBHOOKS=false

COPY --from=build --chown=waapi:waapi /app /app
USER waapi
EXPOSE 3000
HEALTHCHECK --interval=10s --timeout=3s --retries=5 --start-period=20s \
  CMD node -e "fetch('http://127.0.0.1:'+(process.env.API_PORT||3000)+'/health').then(r=>process.exit(r.ok?0:1)).catch(()=>process.exit(1))"
CMD ["node", "apps/api/dist/index.js"]
